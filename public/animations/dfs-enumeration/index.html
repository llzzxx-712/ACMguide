void dfs(int row) {    if (row == n + 1) {        output(); return;    }    for (int c = 1; c <= cols; c++) {        // 本层 c 只负责当前行        // 纯枚举：每一列都可以选        path[row] = c;        // 本模式不记录列、斜线占用        dfs(row + 1);        // 返回本层，接着执行 c++        // 下一次选择会覆盖 path[row]    }}

output() 表示输出当前完整摆法，完整程序见教程。前缀外的旧 path 值不再表示当前选择；四皇后模式还要撤销占用标记。
